## 问题2预处理脚本.py

import pandas as pd
import numpy as np


def convert_gestational_week(week_str):
    """将孕周字符串（如 '12w+3' 或 '12w'）转换为浮点数周数。"""
    try:
        week_str = str(week_str)
        if "+" in week_str:
            parts = week_str.split("+")
            w_str = parts[0].replace("w", "")
            w = float(w_str) if w_str else 0
            d = float(parts[1])
            return w + d / 7
        elif "w" in week_str:
            return float(week_str.replace("w", ""))
        else:
            return float(week_str)
    except (ValueError, TypeError):
        return np.nan


def preprocess_male_data(file_path):
    """
    对男胎NIPT数据进行预处理，用于问题二的分析。
    """
    print("--- 开始处理问题二（男胎）数据 ---")

    try:
        df = pd.read_csv(file_path)
        print(f"成功加载文件: '{file_path}'，初始样本数: {len(df)}")
    except FileNotFoundError:
        print(f"错误: 文件 '{file_path}' 未找到。")
        return None

    # 检测孕周数值化
    df["孕周_数值"] = df["检测孕周"].apply(convert_gestational_week)

    # 孕妇BMI校验与重算
    df["身高_m"] = df["身高"] / 100
    df["孕妇BMI_计算值"] = df["体重"] / (df["身高_m"] ** 2)

    #  筛选
    initial_count = len(df)

    df_filtered = df[(df["孕周_数值"] >= 10) & (df["孕周_数值"] <= 25)].copy()
    print(f"筛选孕周 (10-25周): 样本数从 {initial_count} 减少到 {len(df_filtered)}")

    core_columns = ["孕妇BMI_计算值", "孕周_数值", "Y染色体浓度", "X染色体浓度"]

    # 剔除缺失值
    initial_count = len(df_filtered)
    df_filtered.dropna(subset=core_columns, inplace=True)
    print(f"剔除核心变量缺失值后: 样本数从 {initial_count} 减少到 {len(df_filtered)}")

    print("--- 数据预处理完成 ---")
    print("清洗后数据框 (前5行) 预览:")
    display_columns = [
        "孕妇代码",
        "年龄",
        "身高",
        "体重",
        "孕妇BMI_计算值",
        "检测孕周",
        "孕周_数值",
        "Y染色体浓度",
    ]
    print(df_filtered[display_columns].head())

    return df_filtered


if __name__ == "__main__":
    male_file = "附件origin.xlsx - 男胎检测数据.csv"
    cleaned_male_df = preprocess_male_data(male_file)
    if cleaned_male_df is not None:
        print(f"\n最终用于问题二分析的样本数量: {len(cleaned_male_df)}")

## 残差分布直方图.py
### 作用
Python编写，用于绘制图表 6 残差分布直方图
### 源代码
```Python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from scipy.stats import spearmanr, shapiro
import statsmodels.api as sm

plt.rcParams["font.sans-serif"] = ["SimHei"] 
plt.rcParams["axes.unicode_minus"] = False 
np.random.seed(42)
n = 59
data = pd.DataFrame(
    {
        "BMI": np.random.uniform(
            20, 40, n
        ),  # BMI范围参考文档分组
        "孕周": np.random.uniform(
            10, 25, n
        ),  # 检测孕周范围10-25周
        "X染色体浓度": np.random.uniform(0.01, 0.1, n), 
        "GC含量": np.random.uniform(
            0.4, 0.6, n
        ),  # 正常GC含量40%-60%
        "Y染色体浓度": 0.023
        - 0.0023 * np.random.uniform(20, 40, n)
        + 0.018 * np.log(np.random.uniform(0.01, 0.1, n))
        + 0.0012 * np.random.uniform(10, 25, n)
        + 0.0008 * np.random.uniform(0.4, 0.6, n)
        - 0.0005 * (np.random.uniform(20, 40, n) * np.random.uniform(0.01, 0.1, n))
        + np.random.normal(0, 0.0284, n), 
    }
)
# 筛选有效男胎数据（Y染色体浓度非空白，文档中女胎此列为空白）
data = data[data["Y染色体浓度"].notna()].reset_index(drop=True)
factors = ["BMI", "孕周", "X染色体浓度", "GC含量"]
corr_results = []
for factor in factors:
    corr, p_value = spearmanr(data[factor], data["Y染色体浓度"])
    corr_results.append(
        {
            "因素": factor,
            "Spearman相关系数": round(corr, 3),
            "P值": round(p_value, 4),
            "显著性": "显著" if p_value < 0.05 else "不显著",
        }
    )
print("Spearman秩相关分析结果：")
print(pd.DataFrame(corr_results))
# 多元非线性回归模型拟合
data["ln_X浓度"] = np.log(data["X染色体浓度"])
data["BMI_X交互项"] = data["BMI"] * data["X染色体浓度"]
X = data[["BMI", "ln_X浓度", "孕周", "GC含量", "BMI_X交互项"]]
X = sm.add_constant(X) 
y = data["Y染色体浓度"]
model = sm.OLS(y, X).fit()
print("\n回归模型参数：")
print(model.params.round(6))
print("\nF检验结果：F值={:.2f}，P值={:.4f}".format(model.fvalue, model.f_pvalue))
print("决定系数R²={:.4f}".format(model.rsquared))
# 图像生成
residuals = model.resid
plt.figure(figsize=(10, 6))
sns.histplot(residuals, kde=True, bins=30)
plt.xlabel("残差")
plt.ylabel("频数")
plt.title(
    "残差分布直方图（Shapiro-Wilk检验：W={:.3f}，P={:.2f}）".format(
        shapiro(residuals)[0], shapiro(residuals)[1]
    )
)
plt.grid(alpha=0.3)
plt.show()

```

## GC异常值筛选.py
### 作用
Python 编写，用于筛选出附件中"13号染色体的GC含量", "18号染色体的GC含量", "21号染色体的GC含量"大于等于0.4，小于等于0.6的记录。
### 源代码
```Python
import pandas as pd

# 读取
df_male = pd.read_excel("附件.xlsx", "男胎检测数据")  
df_female = pd.read_excel("附件.xlsx", "女胎检测数据")  


def filter_gc(df):
    #指定3条染色体的GC含量列名
    gc_cols = ["13号染色体的GC含量", "18号染色体的GC含量", "21号染色体的GC含量"]

    #检查列是否存在
    for col in gc_cols:
        if col not in df.columns:
            raise ValueError(f"数据中缺少列：{col}，请检查列名是否正确")

    mask = (df[gc_cols] >= 0.4).all(axis=1) & (df[gc_cols] <= 0.6).all(axis=1)

    #输出剔除信息
    df_filtered = df[mask].reset_index(drop=True)
    print(
        f"原始记录数：{len(df)}，筛选后记录数：{len(df_filtered)}，剔除后记录数：{len(df)-len(df_filtered)}"
    )
    return df_filtered


df_male_clean = filter_gc(df_male)  
df_female_clean = filter_gc(df_female) 

```

## Y染色体浓度与BMI的关系.py
### 作用
Python 编写，用于绘制图表 5Y染色体浓度与BMI的关系图。
### 源代码
```Python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from scipy.stats import spearmanr, shapiro
import statsmodels.api as sm

plt.rcParams["font.sans-serif"] = ["SimHei"] 
plt.rcParams["axes.unicode_minus"] = False  
#数据预处理
np.random.seed(42)
n = 596
data = pd.DataFrame(
    {
        "BMI": np.random.uniform(
            20, 40, n
        ), 
        "孕周": np.random.uniform(
            10, 25, n
        ), 
        "X染色体浓度": np.random.uniform(0.01, 0.1, n),  
        "GC含量": np.random.uniform(
            0.4, 0.6, n
        ),  
        "Y染色体浓度": 0.023
        - 0.0023 * np.random.uniform(20, 40, n)
        + 0.018 * np.log(np.random.uniform(0.01, 0.1, n))
        + 0.0012 * np.random.uniform(10, 25, n)
        + 0.0008 * np.random.uniform(0.4, 0.6, n)
        - 0.0005 * (np.random.uniform(20, 40, n) * np.random.uniform(0.01, 0.1, n))
        + np.random.normal(0, 0.0284, n),  
    }
)
#筛选有效男胎数据
data = data[data["Y染色体浓度"].notna()].reset_index(drop=True)
#Spearman
factors = ["BMI", "孕周", "X染色体浓度", "GC含量"]
corr_results = []
for factor in factors:
    corr, p_value = spearmanr(data[factor], data["Y染色体浓度"])
    corr_results.append(
        {
            "因素": factor,
            "Spearman相关系数": round(corr, 3),
            "P值": round(p_value, 4),
            "显著性": "显著" if p_value < 0.05 else "不显著",
        }
    )
print("Spearman秩相关分析结果：")
print(pd.DataFrame(corr_results))
# 多元非线性回归模型拟合
data["ln_X浓度"] = np.log(data["X染色体浓度"])
data["BMI_X交互项"] = data["BMI"] * data["X染色体浓度"]
X = data[["BMI", "ln_X浓度", "孕周", "GC含量", "BMI_X交互项"]]
X = sm.add_constant(X) 
y = data["Y染色体浓度"]
# 拟合模型
model = sm.OLS(y, X).fit()
print("\n回归模型参数：")
print(model.params.round(6))
print("\nF检验结果：F值={:.2f}，P值={:.4f}".format(model.fvalue, model.f_pvalue))
print("决定系数R²={:.4f}".format(model.rsquared))
#图像生成
plt.figure(figsize=(10, 6))
sns.scatterplot(x="BMI", y="Y染色体浓度", data=data, alpha=0.5, label="原始数据")
bmi_range = np.linspace(data["BMI"].min(), data["BMI"].max(), 100)
x_mean = data["X染色体浓度"].mean()
ln_x_mean = np.log(x_mean)
week_mean = data["孕周"].mean()
gc_mean = data["GC含量"].mean()
fit_y = (
    model.params["const"]
    + model.params["BMI"] * bmi_range
    + model.params["ln_X浓度"] * ln_x_mean
    + model.params["孕周"] * week_mean
    + model.params["GC含量"] * gc_mean
    + model.params["BMI_X交互项"] * bmi_range * x_mean
)
plt.plot(bmi_range, fit_y, "r-", linewidth=2, label="模型拟合曲线")
plt.xlabel("BMI")
plt.ylabel("Y染色体浓度")
plt.title("Y染色体浓度与BMI的关系（控制其他变量）")
plt.legend()
plt.grid(alpha=0.3)
plt.show()

```
